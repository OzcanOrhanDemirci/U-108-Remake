// Saç: daire kümeleri (karşıdan / yandan), aynı sayıda daire → yumuşak dönüş.
// [x, y, r] baş yereli: (0,0) boyun; yüz -0.2 yukarıya uzanır, saç tepesi -0.211
const HAIR_SIDE: [number, number, number][] = [
  [-0.046, -0.163, 0.032], [-0.03, -0.19, 0.034], [0.0, -0.198, 0.036], [0.03, -0.19, 0.032],
  [0.052, -0.172, 0.026], [0.062, -0.152, 0.02], [-0.058, -0.14, 0.024], [-0.052, -0.118, 0.018],
  [0.012, -0.176, 0.03], [-0.016, -0.172, 0.03],
];
const HAIR_FRONT: [number, number, number][] = [
  [-0.04, -0.165, 0.03], [-0.022, -0.19, 0.032], [0.004, -0.198, 0.034], [0.03, -0.188, 0.03],
  [0.048, -0.166, 0.026], [0.056, -0.146, 0.018], [-0.054, -0.146, 0.02], [-0.056, -0.126, 0.014],
  [0.016, -0.174, 0.03], [-0.014, -0.172, 0.03],
];

function drawHead(ctx: Ctx, headTurn: number, F: Face, C: (c: string) => string, look: number) {
  const t = Math.abs(headTurn);
  const dir = headTurn < 0 ? -1 : 1;
  ctx.save();
  if (dir < 0) ctx.scale(-1, 1);
  const fw = lerp(0.082, 0.078, t);       // yüz genişliği
  const fx = lerp(0, 0.006, t);           // yüz merkezi kayması
  const faceTop = -0.19, faceBot = 0.004;
  const ly = look * 0.006;
  // arka saç (kafanın arkasından taşan bukleler)
  ctx.fillStyle = C(PAL.sacKoyu);
  for (let i = 0; i < HAIR_SIDE.length; i++) {
    const a = HAIR_FRONT[i], b = HAIR_SIDE[i];
    const x = lerp(a[0], b[0], t), y = lerp(a[1], b[1], t), r = lerp(a[2], b[2], t);
    ctx.beginPath(); ctx.arc(x - 0.004, y + 0.004, r, 0, TAU); ctx.fill();
  }
  // kulak (arka taraf)
  const earX = lerp(-fw / 2 - 0.006, -fw / 2 + 0.004, t) + fx;
  ctx.fillStyle = C(PAL.yanak);
  ctx.beginPath(); ctx.ellipse(earX, -0.098, 0.012, 0.024, -0.25, 0, TAU); ctx.fill();
  if (t < 0.6) {
    ctx.globalAlpha = 1 - t / 0.6;
    ctx.beginPath(); ctx.ellipse(fw / 2 + 0.006 + fx, -0.098, 0.012, 0.024, 0.25, 0, TAU); ctx.fill();
    ctx.globalAlpha = 1;
  }
  // yüz dikdörtgeni
  ctx.fillStyle = C(PAL.ten);
  roundRect(ctx, -fw / 2 + fx, faceTop, fw, faceBot - faceTop, 0.012);
  ctx.fill();
  // sakal çizgileri (çene V'si) ve bıyık
  const cx = fx + lerp(0, 0.012, t);
  ctx.strokeStyle = C(PAL.sakal); ctx.lineCap = 'round';
  ctx.lineWidth = 0.0055;
  ctx.beginPath();
  ctx.moveTo(-fw / 2 + fx + 0.004, -0.062); ctx.lineTo(cx - 0.012, 0.002);
  ctx.moveTo(fw / 2 + fx - 0.004, -0.062); ctx.lineTo(cx + 0.012, 0.002);
  ctx.stroke();
  ctx.lineWidth = 0.011;
  ctx.beginPath();
  ctx.moveTo(-fw / 2 + fx + 0.003, -0.07);
  ctx.quadraticCurveTo(cx - 0.012, -0.05 + ly, cx, -0.054 + ly);
  ctx.quadraticCurveTo(cx + 0.014, -0.05 + ly, fw / 2 + fx - 0.003, -0.07);
  ctx.stroke();
  // ağız
  const mo = F.mouthOpen, sm = F.smile;
  ctx.fillStyle = C(PAL.dudak);
  const my = -0.036 + ly;
  if (Math.abs(sm) > 0.25 && mo < 0.3) {
    ctx.strokeStyle = C(PAL.dudak); ctx.lineWidth = 0.006;
    ctx.beginPath(); ctx.moveTo(cx - 0.012, my - sm * 0.002); ctx.quadraticCurveTo(cx, my + sm * 0.01, cx + 0.012, my - sm * 0.002); ctx.stroke();
  } else {
    ctx.beginPath(); ctx.ellipse(cx + 0.001, my + mo * 0.003, 0.0105 - mo * 0.002, 0.0042 + mo * 0.008, 0, 0, TAU); ctx.fill();
    if (mo > 0.2) { ctx.fillStyle = C('#5a1a14'); ctx.beginPath(); ctx.ellipse(cx + 0.001, my + mo * 0.004, 0.006, 0.002 + mo * 0.005, 0, 0, TAU); ctx.fill(); }
  }
  // burun
  const nx = cx + lerp(0, 0.002, t);
  const ng = ctx.createLinearGradient(0, -0.1 + ly, 0, -0.06 + ly);
  ng.addColorStop(0, C(PAL.burunKoyu)); ng.addColorStop(1, C(PAL.burun));
  ctx.fillStyle = ng;
  ctx.beginPath();
  ctx.moveTo(nx - 0.004, -0.1 + ly);
  ctx.lineTo(nx + 0.004, -0.1 + ly);
  ctx.quadraticCurveTo(nx + 0.006, -0.078 + ly, nx + 0.015, -0.07 + ly);
  ctx.quadraticCurveTo(nx + 0.016, -0.06 + ly, nx, -0.06 + ly);
  ctx.quadraticCurveTo(nx - 0.016, -0.06 + ly, nx - 0.015, -0.07 + ly);
  ctx.quadraticCurveTo(nx - 0.006, -0.078 + ly, nx - 0.004, -0.1 + ly);
  ctx.closePath(); ctx.fill();
  ctx.fillStyle = C('#9a4a3a');
  ctx.beginPath(); ctx.ellipse(nx - 0.006, -0.064 + ly, 0.0022, 0.0034, 0, 0, TAU); ctx.ellipse(nx + 0.006, -0.064 + ly, 0.0022, 0.0034, 0, 0, TAU); ctx.fill();
  // allık
  ctx.fillStyle = C(PAL.allik); ctx.globalAlpha = 0.85;
  const e1x = cx - lerp(0.022, 0.019, t), e2x = cx + lerp(0.022, 0.024, t);
  const ey = -0.104 + ly;
  ctx.beginPath(); ctx.ellipse(e1x, ey + 0.012, 0.009, 0.0035, 0, 0, TAU); ctx.ellipse(e2x, ey + 0.012, 0.009, 0.0035, 0, 0, TAU); ctx.fill();
  ctx.globalAlpha = 1;
  // gözler
  const blink = clamp(F.blink, 0, 1);
  const ex = F.eyeX * 0.004, eyy = F.eyeY * 0.003;
  for (const x of [e1x, e2x]) {
    if (blink > 0.7) {
      ctx.strokeStyle = C(PAL.goz); ctx.lineWidth = 0.0028;
      ctx.beginPath(); ctx.moveTo(x - 0.005, ey); ctx.quadraticCurveTo(x, ey + 0.003, x + 0.005, ey); ctx.stroke();
    } else {
      ctx.fillStyle = C(PAL.goz);
      ctx.beginPath(); ctx.ellipse(x + ex, ey + eyy, 0.0042, 0.0058 * (1 - blink), 0, 0, TAU); ctx.fill();
      if (F.tear > 0) {
        ctx.fillStyle = `rgba(255,255,255,${0.9 * F.tear})`;
        ctx.beginPath(); ctx.arc(x + ex + 0.0014, ey + eyy - 0.002, 0.0016, 0, TAU); ctx.fill();
      }
    }
  }
  // gözlük
  ctx.strokeStyle = C(PAL.gozluk); ctx.lineWidth = 0.0024;
  const gr = 0.0235;
  ctx.beginPath(); ctx.arc(e1x + 0.001, ey + 0.001, gr, 0, TAU); ctx.stroke();
  ctx.beginPath(); ctx.arc(e2x - 0.001, ey + 0.001, gr, 0, TAU); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(e1x + gr, ey - 0.002); ctx.quadraticCurveTo((e1x + e2x) / 2, ey - 0.006, e2x - gr, ey - 0.002); ctx.stroke();
  // camda ışıltı
  ctx.strokeStyle = 'rgba(255,255,255,0.35)'; ctx.lineWidth = 0.002;
  ctx.beginPath(); ctx.arc(e1x, ey, gr * 0.72, -2.4, -1.7); ctx.stroke();
  ctx.beginPath(); ctx.arc(e2x, ey, gr * 0.72, -2.4, -1.7); ctx.stroke();
  // kaşlar
  ctx.strokeStyle = C(PAL.kas); ctx.lineWidth = 0.0085; ctx.lineCap = 'round';
  const by = ey - 0.032 - F.browUp * 0.007;
  for (const [x, s] of [[e1x, -1], [e2x, 1]] as [number, number][]) {
    const inner = x - s * 0.012, outer = x + s * 0.014;
    const iy = by - F.browSad * 0.008 + (F.browUp < 0 ? -F.browUp * 0.005 : 0);
    const oy = by + F.browSad * 0.003;
    ctx.beginPath(); ctx.moveTo(inner, iy); ctx.quadraticCurveTo(x, by - 0.004, outer, oy); ctx.stroke();
  }
  // saç (ön kütle)
  const hairShapes = HAIR_SIDE.map((b, i) => { const a = HAIR_FRONT[i]; return [lerp(a[0], b[0], t), lerp(a[1], b[1], t), lerp(a[2], b[2], t)] as [number, number, number]; });
  ctx.fillStyle = C(PAL.sac);
  ctx.beginPath();
  for (const [x, y, r] of hairShapes) { ctx.moveTo(x + r, y); ctx.arc(x, y, r, 0, TAU); }
  ctx.fill();
  // alın üstü kahkül hattı: yüzün üst kenarını saçla kapat
  ctx.beginPath();
  ctx.moveTo(-fw / 2 + fx - 0.004, faceTop + 0.012);
  ctx.quadraticCurveTo(fx - 0.01, faceTop + 0.04, fx + 0.006, faceTop + 0.028);
  ctx.quadraticCurveTo(fx + 0.024, faceTop + 0.02, fw / 2 + fx + 0.006, faceTop + 0.046);
  ctx.lineTo(fw / 2 + fx + 0.006, faceTop - 0.01); ctx.lineTo(-fw / 2 + fx - 0.004, faceTop - 0.01);
  ctx.closePath(); ctx.fill();
  // saç parlaklığı
  ctx.fillStyle = C(PAL.sacAcik); ctx.globalAlpha = 0.75;
  ctx.beginPath();
  ctx.moveTo(-0.03, -0.205); ctx.quadraticCurveTo(0.01, -0.222, 0.045, -0.198);
  ctx.quadraticCurveTo(0.012, -0.21, -0.03, -0.198); ctx.closePath(); ctx.fill();
  ctx.beginPath(); ctx.arc(0.058, -0.168, 0.012, -1.2, 1.2); ctx.lineTo(0.05, -0.16); ctx.fill();
  ctx.globalAlpha = 1;
  ctx.restore();
}
